# Each suite's main() is renamed <suite>_main, and its prototype comes from a
# generated suites.h, force-included, so the suites compile unchanged.
get_filename_component(TTEC_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${TTEC_ROOT}/tests/portable.cmake")

set(srcs "run_tests.c")
set(protos "")
set(xlist "")
foreach(t IN LISTS TTEC_PORTABLE_TESTS)
    list(APPEND srcs "${TTEC_ROOT}/tests/${t}.c")
    string(APPEND protos "int ${t}_main(void);\n")
    string(APPEND xlist " X(${t})")
endforeach()

idf_component_register(SRCS ${srcs}
                       INCLUDE_DIRS "." "${TTEC_ROOT}/tests"
                       PRIV_REQUIRES ttec esp_system)

file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/suites.h"
     "/* Generated by ground/esp32-tests/main/CMakeLists.txt. */\n"
     "#ifndef SUITES_H\n#define SUITES_H\n${protos}"
     "#define TTEC_SUITES(X)${xlist}\nvoid app_main(void);\n#endif\n")
target_include_directories(${COMPONENT_LIB} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
# The suites with the project's warnings; run_tests.c, which includes
# FreeRTOS, with IDF's.
foreach(t IN LISTS TTEC_PORTABLE_TESTS)
    set_source_files_properties("${TTEC_ROOT}/tests/${t}.c" PROPERTIES
        COMPILE_DEFINITIONS "main=${t}_main"
        COMPILE_OPTIONS "-include;${CMAKE_CURRENT_BINARY_DIR}/suites.h;-std=c11;-Wall;-Wextra;-Wshadow;-Wconversion;-Wsign-conversion;-Wstrict-prototypes;-Wmissing-prototypes;-Wsign-compare;-Wunused-parameter;-Werror")
endforeach()

idf_build_get_property(idf_path IDF_PATH)
target_include_directories(${COMPONENT_LIB} SYSTEM PRIVATE "${idf_path}/components/newlib/platform_include")
